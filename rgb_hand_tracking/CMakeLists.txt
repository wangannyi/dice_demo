cmake_minimum_required(VERSION 3.16)
project(rgb_hand_landmark_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(OpenCV REQUIRED COMPONENTS core imgproc imgcodecs videoio)
set(ONNXRUNTIME_ROOT /usr/local CACHE PATH "ONNX Runtime prefix")
find_path(ORT_INCLUDE onnxruntime_cxx_api.h PATHS "${ONNXRUNTIME_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(ORT_LIB onnxruntime PATHS "${ONNXRUNTIME_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
add_executable(landmark_bridge landmark_bridge.cpp vendor/hand2/src/hand_pipeline.cpp)
target_include_directories(landmark_bridge PRIVATE vendor/hand2/src "${ORT_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(landmark_bridge PRIVATE "${ORT_LIB}" ${OpenCV_LIBS})
find_package(Threads REQUIRED)
add_executable(no_marker_live no_marker_live.cpp vendor/hand2/src/hand_pipeline.cpp)
target_include_directories(no_marker_live PRIVATE vendor/hand2/src "${ORT_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(no_marker_live PRIVATE "${ORT_LIB}" ${OpenCV_LIBS} Threads::Threads)
add_executable(no_marker_robot_live no_marker_live.cpp robot_hand2/hand_pipeline.cpp)
target_compile_definitions(no_marker_robot_live PRIVATE HAND2_ROBOT_PIPELINE=1)
target_include_directories(no_marker_robot_live PRIVATE robot_hand2 "${ORT_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(no_marker_robot_live PRIVATE "${ORT_LIB}" ${OpenCV_LIBS} Threads::Threads)
add_executable(no_marker_replay no_marker_replay.cpp vendor/hand2/src/hand_pipeline.cpp)
target_include_directories(no_marker_replay PRIVATE vendor/hand2/src "${ORT_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(no_marker_replay PRIVATE "${ORT_LIB}" ${OpenCV_LIBS})
add_executable(no_marker_robot_replay no_marker_replay.cpp robot_hand2/hand_pipeline.cpp)
target_compile_definitions(no_marker_robot_replay PRIVATE HAND2_ROBOT_PIPELINE=1)
target_include_directories(no_marker_robot_replay PRIVATE robot_hand2 "${ORT_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(no_marker_robot_replay PRIVATE "${ORT_LIB}" ${OpenCV_LIBS})
add_executable(hand_observation_bridge hand_observation_bridge.cpp robot_hand2/hand_pipeline.cpp)
target_include_directories(hand_observation_bridge PRIVATE robot_hand2 "${ORT_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(hand_observation_bridge PRIVATE "${ORT_LIB}" ${OpenCV_LIBS})
